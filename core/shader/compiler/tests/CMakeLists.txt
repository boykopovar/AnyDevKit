add_executable(shader_compilation_tests CompileStages.cpp)
target_link_libraries(shader_compilation_tests PRIVATE AnyDevKit::ShaderCompiler)
target_compile_features(shader_compilation_tests PRIVATE cxx_std_20)
set_target_properties(shader_compilation_tests PROPERTIES CXX_EXTENSIONS OFF)
if(MINGW)
    target_link_options(shader_compilation_tests PRIVATE -static -static-libgcc -static-libstdc++)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_link_options(shader_compilation_tests PRIVATE -fuse-ld=lld)
    endif()
endif()
foreach(stage IN ITEMS vertex fragment compute geometry tessellation_control tessellation_evaluation export local)
    add_test(NAME shader_compile_${stage} COMMAND shader_compilation_tests "${stage}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders")
    set_tests_properties(shader_compile_${stage} PROPERTIES TIMEOUT 60)
endforeach()
